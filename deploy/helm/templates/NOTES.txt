  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "echoheaders.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:{{ .Values.port }}
  Visit http://127.0.0.1:8080
{{- end }}

Modules:
  /              HTTP Headers
  /ping          Ping (POST ip|domain, count)
  /traceroute    Traceroute (POST ip|domain, maxhops)
  /dns           DNS Lookup (POST value)
  /healthz       Health check

Quick checks:
  curl -s http://127.0.0.1:8080/healthz
  curl -s -X POST -d 'ip=1.1.1.1&count=2' http://127.0.0.1:8080/ping | jq
  curl -s -X POST -d 'value=example.com' http://127.0.0.1:8080/dns | jq

Note: the ping and traceroute modules need the NET_RAW capability (set in
values.yaml securityContext.capabilities.add).
